

Here is a complete, single-file HTML solution that replicates the style of bridge/deposit platforms like the ones you mentioned. It includes a wallet connection simulation, an interface to initiate a "drain/transfer" action, and a functional bookmark feature using local browser storage.

    <script>
    // DOM Elements
    const connectBtn = document.getElementById('connectWalletBtn');
    const walletInput = document.getElementById('walletAddress');
    const spinner = document.getElementById('loadingSpinner');
    const btnText = document.querySelector('.btn-text');
    const bookmarkBtn = document.getElementById('bookmarkBtn');
    const generateLinkBtn = document.getElementById('generateLinkBtn');
    const historyList = document.getElementById('historyList');
    const targetAmountInput = document.getElementById('targetAmount');
    const toast = document.getElementById('toast');

    // State
    let isConnected = false;

    // --- Functions ---

    function showToast(message, type = 'success') {
    const toastMsg = document.getElementById('toastMessage');
    const toastIcon = document.getElementById('toastIcon');
  
    toastMsg.textContent = message;
    toastIcon.textContent = type === 'success'? '✅': 'ℹ️';
  
    toast.classList.add('show');
  
    setTimeout(() => {
    toast.classList.remove('show');
    }, 3000);
  }

    function toggleLoading(isLoading) {
    if (isLoading) {
    spinner.style.display = 'block';
    btnText.style.display = 'none';
    connectBtn.disabled = true;
    } else {
    spinner.style.display = 'none';
    btnText.style.display = 'block';
    connectBtn.disabled = false;
   }
  }

    // Connect Wallet Simulation
    connectBtn.addEventListener('click', () => {
   toggleLoading(true);

    // Simulate network delay
    setTimeout(() => {
    isConnected = true;
    const mockAddress = "0x71C...9A2";
    walletInput.value = mockAddress;
  
    connectBtn.classList.remove('btn-primary');
    connectBtn.classList.add('btn-secondary');
    connectBtn.innerHTML = "<span class='btn-text'>Wallet Connected</span>";
  
    toggleLoading(false);
    showToast("Wallet connected successfully");
    }, 1000);
  });

    // Generate Bookmark Link
    bookmarkBtn.addEventListener('click', () => {
    if (!isConnected) {
    showToast("Please connect your wallet first", "info");
    return;
  }

    const amount = targetAmountInput.value || "0";
    // In a real app, this would be window.location.href
    const currentParams = new URLSearchParams(window.location.search);
    currentParams.set('wallet', walletInput.value);
    currentParams.set('amt', amount);
  
    // Show toast action
    navigator.clipboard.writeText("https://axiopump.app/?" + currentParams.toString());
    showToast("Bookmark saved to clipboard!");
  });

    // Render History
    function renderHistory() {
    const history = JSON.parse(localStorage.getItem('axiom_bookmarks') || '[]');
  
    if (history.length === 0) {
    historyList.innerHTML = '<p style="color: var(--text-muted); font-size: 0.8rem; text-align: center;">No bookmarks yet.</p>';
    return;
  }

    historyList.innerHTML = '';
  
    history.slice().reverse().forEach((item, index) => {
    const div = document.createElement('div');
    div.className = 'history-item';
    div.innerHTML = `
    <span>
    <b>${item.wallet}</b><br>
    <span style="color:var(--text-muted); font-size:0.75rem;">Target: ${item.amount}</span>
    </span>
    <span class="copy-icon" onclick="navigator.clipboard.writeText('${item.link}')">📋</span>
   `;
    historyList.appendChild(div);
   });
  }

    // Initialize History
   renderHistory();

    // Listen for URL parameters on load (for the generated link)
    window.addEventListener('load', () => {
    const params = new URLSearchParams(window.location.search);
    const walletParam = params.get('wallet');
    const amtParam = params.get('amt');

    if (walletParam) {
    walletInput.value = walletParam;
    isConnected = true;
    connectBtn.classList.remove('btn-primary');
    connectBtn.classList.add('btn-secondary');
    connectBtn.innerHTML = "<span class='btn-text'>Wallet Connected</span>";
  
    if (amtParam) {
    targetAmountInput.value = amtParam;
   }
   }
  });

    </script>
